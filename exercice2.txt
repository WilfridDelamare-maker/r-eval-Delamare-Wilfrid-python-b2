from fastapi import Depends, FastAPI, HTTPException
from app.models import Station, StationPatch
from app.db import get_db, init_db, SessionLocal
from sqlalchemy.orm import Session

app = FastAPI()

STATIONS = []

@app.get('/health')
def health():
    return {"status": "ok"}

@app.post("/stations", status_code=201)
def post_stations(station: Station):
    for s in STATIONS:
        if station.code == s["code"]:
            raise HTTPException(status_code=409, detail="code already existing")

    new_station = {
        "id": len(STATIONS) + 1,
        "code": station.code,
        "name": station.name,
        "capacity": station.capacity,
        "status": station.status,
    }
    STATIONS.append(new_station)
    return new_station

@app.get('/stations')
def get_station(status: str | None = None):
    if status is None:
        return STATIONS

    return [station for station in STATIONS if station["status"] == status]

@app.get("/stations/{station_id}")
def get_station_by_id(station_id: int):
    for s in STATIONS:
        if s["id"] == station_id:
            return s
    raise HTTPException(status_code=404, detail="Station not found")

@app.patch("/stations/{station_id}")
def patch_station(station_id: int, data: StationPatch):
    for station in STATIONS:
        if station["id"] == station_id:
            change = data.model_dump(exclude_unset=True)
            station.update(change)
            return station

    raise HTTPException(status_code=404, detail="No station to patch")
